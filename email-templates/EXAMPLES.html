🟨 Node.js / JavaScript (Express + Nodemailer)

Instalação:

bash npm install nodemailer

Implementação:

javascript const nodemailer = require('nodemailer'); const fs = require('fs'); const path = require('path'); // Configurar transporter const transporter = nodemailer.createTransport({ service: 'gmail', auth: { user: process.env.EMAIL_USER, pass: process.env.EMAIL_PASS, }, }); // Função para enviar email de verificação async function sendVerificationEmail(userEmail, verificationCode) { const templatePath = path.join(__dirname, 'templates/verification-banner.html'); let emailHtml = fs.readFileSync(templatePath, 'utf-8'); // Substituir placeholder emailHtml = emailHtml.replace('{{CODE}}', verificationCode); await transporter.sendMail({ from: 'noreply@armazix.com.br', to: userEmail, subject: 'Verifique seu email - Armazix', html: emailHtml, }); } // Função para enviar email de reset async function sendPasswordResetEmail(userEmail, resetCode) { const templatePath = path.join(__dirname, 'templates/password-reset-banner.html'); let emailHtml = fs.readFileSync(templatePath, 'utf-8'); // Substituir placeholder emailHtml = emailHtml.replace('{{CODE}}', resetCode); await transporter.sendMail({ from: 'noreply@armazix.com.br', to: userEmail, subject: 'Recuperação de senha - Armazix', html: emailHtml, }); } // Uso: // await sendVerificationEmail('user@example.com', '483921'); // await sendPasswordResetEmail('user@example.com', '728154');
⚠️ Importante: Armazene as credenciais de email em variáveis de ambiente (.env), nunca no código-fonte.

🐍 Python (Flask + Flask-Mail)

Instalação:

bash pip install flask-mail

Implementação:

python from flask import Flask from flask_mail import Mail, Message import os app = Flask(__name__) app.config['MAIL_SERVER'] = os.getenv('MAIL_SERVER') app.config['MAIL_PORT'] = 587 app.config['MAIL_USE_TLS'] = True app.config['MAIL_USERNAME'] = os.getenv('MAIL_USERNAME') app.config['MAIL_PASSWORD'] = os.getenv('MAIL_PASSWORD') mail = Mail(app) def send_verification_email(user_email, code): with open('templates/verification-banner.html', 'r', encoding='utf-8') as f: email_html = f.read() # Substituir placeholder email_html = email_html.replace('{{CODE}}', code) msg = Message( 'Verifique seu email - Armazix', recipients=[user_email], html=email_html, sender='noreply@armazix.com.br' ) mail.send(msg) def send_password_reset_email(user_email, code): with open('templates/password-reset-banner.html', 'r', encoding='utf-8') as f: email_html = f.read() # Substituir placeholder email_html = email_html.replace('{{CODE}}', code) msg = Message( 'Recuperação de senha - Armazix', recipients=[user_email], html=email_html, sender='noreply@armazix.com.br' ) mail.send(msg) # Uso: # send_verification_email('user@example.com', '483921') # send_password_reset_email('user@example.com', '728154')

🐘 PHP (PHPMailer)

Instalação:

bash composer require phpmailer/phpmailer

Implementação:

php <?php use PHPMailer\PHPMailer\PHPMailer; use PHPMailer\PHPMailer\Exception; require 'vendor/autoload.php'; function sendVerificationEmail($userEmail, $code) { $mail = new PHPMailer(true); try { $mail->isSMTP(); $mail->Host = $_ENV['MAIL_HOST']; $mail->SMTPAuth = true; $mail->Username = $_ENV['MAIL_USERNAME']; $mail->Password = $_ENV['MAIL_PASSWORD']; $mail->SMTPSecure = PHPMailer::ENCRYPTION_STARTTLS; $mail->Port = 587; $mail->setFrom('noreply@armazix.com.br', 'Armazix'); $mail->addAddress($userEmail); // Ler template $emailHtml = file_get_contents('templates/verification-banner.html'); $emailHtml = str_replace('{{CODE}}', $code, $emailHtml); $mail->isHTML(true); $mail->Subject = 'Verifique seu email - Armazix'; $mail->Body = $emailHtml; $mail->send(); } catch (Exception $e) { error_log("Email error: {$mail->ErrorInfo}"); } } function sendPasswordResetEmail($userEmail, $code) { $mail = new PHPMailer(true); try { $mail->isSMTP(); $mail->Host = $_ENV['MAIL_HOST']; $mail->SMTPAuth = true; $mail->Username = $_ENV['MAIL_USERNAME']; $mail->Password = $_ENV['MAIL_PASSWORD']; $mail->SMTPSecure = PHPMailer::ENCRYPTION_STARTTLS; $mail->Port = 587; $mail->setFrom('noreply@armazix.com.br', 'Armazix'); $mail->addAddress($userEmail); // Ler template $emailHtml = file_get_contents('templates/password-reset-banner.html'); $emailHtml = str_replace('{{CODE}}', $code, $emailHtml); $mail->isHTML(true); $mail->Subject = 'Recuperação de senha - Armazix'; $mail->Body = $emailHtml; $mail->send(); } catch (Exception $e) { error_log("Email error: {$mail->ErrorInfo}"); } } // Uso: // sendVerificationEmail('user@example.com', '483921'); // sendPasswordResetEmail('user@example.com', '728154');

✅ Boas Práticas

1. Geração de Código

javascript // ✓ BOM - Código aleatório seguro function generateCode() { return Math.floor(100000 + Math.random() * 900000).toString(); } // ✓ BOM - Com caracteres alfanuméricos function generateAlphanumericCode() { return Math.random().toString(36).substring(2, 8).toUpperCase(); }

2. Armazenamento Seguro

  • Armazene hash do código, não o código em texto plano
  • Use salt para hash (bcrypt, PBKDF2, argon2)
  • Defina expiração (15-30 minutos é ideal)
  • Limite tentativas (máximo 5 por código)
  • Limpe códigos expirados regularmente

3. Segurança Email

  • Use TLS/SSL para SMTP
  • Implemente DKIM, SPF, DMARC
  • Configure Reply-To diferente de From (evita spam)
  • Use endereço genérico (noreply@)
  • Nunca responda aos emails automáticos

4. UX de Email

  • Assunto claro e descritivo
  • Inclua link além do código (opção)
  • Mostre data/hora de expiração
  • Aviso se código foi solicitado por eles
  • Link para cancelar se não solicitou

🔧 Resolvendo Problemas

Email não chega:

  • Verifique se credenciais SMTP estão corretas
  • Tente usar porta 25 ou 465 se 587 não funcionar
  • Verifique whitelist/blacklist do servidor
  • Implemente SPF, DKIM e DMARC records no DNS
  • Cheque logs do servidor de email

Design quebrado no cliente:

  • Alguns clientes não suportam CSS inline complexo
  • Testar com fallbacks para clientes antigos
  • Use tabelas como backup se necessário
  • Verifique media queries funcionam (mobile)
  • Teste dark mode em todos os clientes

Placeholder não substitui:

  • Verifique exatamente {{CODE}} (sem espaços extras)
  • Use .replace() simples, não regex complexo
  • Garanta que arquivo é lido antes de substituir
  • Teste em desenvolvimento primeiro
  • Verifique encoding UTF-8 do arquivo
